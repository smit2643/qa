# Bug0.com Clone — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a full SaaS clone of bug0.com — an AI-powered E2E testing platform that generates, runs, self-heals, and reports on Playwright browser tests continuously.

**Architecture:** Modular FastAPI monolith (Python) + separate Celery-based test runner (Python). Frontend in Next.js 14. browser-use handles the AI↔browser layer. All services containerized via Docker Compose.

**Tech Stack:** Next.js 14, Tailwind CSS, shadcn/ui, FastAPI, SQLAlchemy 2.0, Alembic, PostgreSQL, Redis, Celery, playwright-python, browser-use, Claude 3.7 Sonnet (default LLM), MinIO, Stripe, SendGrid, Nginx, Docker.

# backend/modules/auth/service.py
from datetime import datetime, timedelta
from jose import jwt
from passlib.context import CryptContext
from sqlalchemy.orm import Session
from models import User, Organization, Membership, Role
from core.config import settings
import uuid

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
ALGORITHM = "HS256"

def hash_password(password: str) -> str:
    return pwd_context.hash(password)

def verify_password(plain: str, hashed: str) -> bool:
    return pwd_context.verify(plain, hashed)

def create_access_token(user_id: str) -> str:
    expire = datetime.utcnow() + timedelta(minutes=settings.access_token_expire_minutes)
    return jwt.encode({"sub": user_id, "exp": expire}, settings.secret_key, algorithm=ALGORITHM)

def decode_token(token: str) -> str:
    payload = jwt.decode(token, settings.secret_key, algorithms=[ALGORITHM])
    return payload["sub"]

def signup(db: Session, email: str, name: str, password: str) -> User:
    existing = db.query(User).filter(User.email == email).first()
    if existing:
        raise ValueError("Email already registered")
    user = User(email=email, name=name, hashed_password=hash_password(password))
    db.add(user)
    # Create personal org
    slug = name.lower().replace(" ", "-") + "-" + str(uuid.uuid4())[:8]
    org = Organization(name=f"{name}'s Org", slug=slug)
    db.add(org)
    db.flush()
    membership = Membership(user_id=user.id, organization_id=org.id, role=Role.owner)
    db.add(membership)
    db.commit()
    db.refresh(user)
    return user

def login(db: Session, email: str, password: str) -> User:
    user = db.query(User).filter(User.email == email).first()
    if not user or not verify_password(password, user.hashed_password or ""):
        raise ValueError("Invalid credentials")
    return user
```

# backend/modules/ai/provider.py
from abc import ABC, abstractmethod
from anthropic import Anthropic
from core.config import settings

class LLMProvider(ABC):
    @abstractmethod
    def complete(self, system: str, user: str) -> str:
        pass

class ClaudeProvider(LLMProvider):
    def __init__(self):
        self.client = Anthropic(api_key=settings.anthropic_api_key)

    def complete(self, system: str, user: str) -> str:
        message = self.client.messages.create(
            model="claude-sonnet-4-6",
            max_tokens=4096,
            system=system,
            messages=[{"role": "user", "content": user}],
        )
        return message.content[0].text

def get_provider() -> LLMProvider:
    return ClaudeProvider()
```

async def get_accessibility_tree(url: str) -> str:
    """Use browser-use agent to visit URL and return accessibility tree snapshot."""
    browser = Browser(config=BrowserConfig(headless=True))
    try:
        async with await browser.new_context() as context:
            page = await context.new_page()
            await page.goto(url)
            await page.wait_for_load_state("networkidle")
            snapshot = await page.accessibility.snapshot()
            return str(snapshot)
    finally:
        await browser.close()

# runner/executor/playwright_executor.py
import asyncio
import traceback
from playwright.async_api import async_playwright, Page

class ExecutionResult:
    def __init__(self, passed: bool, error: str | None, console_logs: list[str]):
        self.passed = passed
        self.error = error
        self.console_logs = console_logs

async def execute_test(code: str, target_url: str, video_dir: str) -> ExecutionResult:
    """Execute a playwright test function defined as `async def run_test(page)`."""
    console_logs: list[str] = []
    error: str | None = None
    passed = False

    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=True)
        context = await browser.new_context(
            record_video_dir=video_dir,
        )
        page: Page = await context.new_page()
        page.on("console", lambda msg: console_logs.append(f"[{msg.type}] {msg.text}"))

        try:
            # Execute the generated test code
            namespace: dict = {}
            exec(compile(code, "<test>", "exec"), namespace)
            run_test = namespace.get("run_test")
            if not run_test:
                raise ValueError("Generated code must define `async def run_test(page)`")
            await run_test(page)
            passed = True
        except Exception as e:
            error = traceback.format_exc()
        finally:
            await context.close()
            await browser.close()

    return ExecutionResult(passed=passed, error=error, console_logs=console_logs)
```

import os, json, tempfile, httpx
from celery_app import app
from executor.playwright_executor import execute_test
from executor.recorder import get_video_path
from storage.artifact_uploader import upload_artifact

BACKEND_URL = os.environ.get("BACKEND_URL", "http://backend:8000")

@app.task(bind=True, max_retries=1)
def run_test_task(self, payload: dict):
    """
    payload: {
        run_id, result_id, test_id, test_code, target_url
    }
    """
    run_id = payload["run_id"]
    result_id = payload["result_id"]
    test_code = payload["test_code"]
    target_url = payload["target_url"]

    with tempfile.TemporaryDirectory() as tmpdir:
        import asyncio
        result = asyncio.run(execute_test(test_code, target_url, tmpdir))

        video_url = None
        video_path = get_video_path(tmpdir)
        if video_path:
            key = f"runs/{run_id}/{result_id}/video.webm"
            video_url = upload_artifact(video_path, key)

        # Write console logs
        log_url = None
        log_path = os.path.join(tmpdir, "console.log")
        with open(log_path, "w") as f:
            f.write("\n".join(result.console_logs))
        log_key = f"runs/{run_id}/{result_id}/console.log"
        log_url = upload_artifact(log_path, log_key)

        # Post result back to backend
        status = "passed" if result.passed else "failed"
        httpx.patch(
            f"{BACKEND_URL}/api/v1/runs/{run_id}/results/{result_id}",
            json={
                "status": status,
                "error_message": result.error,
                "video_url": video_url,
                "log_url": log_url,
            },
            timeout=10.0,
        )

# backend/modules/jobs/service.py
import json
from datetime import datetime
from sqlalchemy.orm import Session
from models import TestRun, TestResult, TestCase, TestSuite, RunStatus, TriggerType, ResultStatus
from core.redis_client import get_redis

CELERY_QUEUE = "celery"

def trigger_run(
    db: Session,
    suite_id: str,
    trigger: TriggerType,
    branch: str | None,
    commit_sha: str | None,
) -> TestRun:
    suite = db.query(TestSuite).filter(TestSuite.id == suite_id).first()
    if not suite:
        raise ValueError(f"Suite {suite_id} not found")

    run = TestRun(
        suite_id=suite_id,
        trigger=trigger,
        branch=branch,
        commit_sha=commit_sha,
        status=RunStatus.queued,
    )
    db.add(run)
    db.flush()

    tests = db.query(TestCase).filter(TestCase.suite_id == suite_id).all()
    for test in tests:
        result = TestResult(run_id=run.id, test_id=test.id, status=ResultStatus.failed)
        db.add(result)
        db.flush()

        # Push Celery task to Redis
        payload = {
            "run_id": run.id,
            "result_id": result.id,
            "test_id": test.id,
            "test_code": test.code,
            "target_url": suite.project.target_url,
        }
        redis = get_redis()
        task = {
            "id": result.id,
            "task": "workers.test_runner.run_test_task",
            "args": [payload],
            "kwargs": {},
            "retries": 0,
        }
        redis.lpush(CELERY_QUEUE, json.dumps(task))

    run.status = RunStatus.running
    run.started_at = datetime.utcnow()
    db.commit()
    db.refresh(run)
    return run

def update_result(
    db: Session,
    run_id: str,
    result_id: str,
    status: ResultStatus,
    **kwargs,
) -> TestResult:
    result = db.query(TestResult).filter(
        TestResult.id == result_id,
        TestResult.run_id == run_id
    ).first()
    if not result:
        raise ValueError("Result not found")

    result.status = status
    for key, value in kwargs.items():
        if value is not None:
            setattr(result, key, value)

    db.commit()

    # Check if all results for the run are done
    run = db.query(TestRun).filter(TestRun.id == run_id).first()
    all_results = db.query(TestResult).filter(TestResult.run_id == run_id).all()
    pending = [r for r in all_results if r.status == ResultStatus.failed and r.video_url is None]
    if not pending:
        all_passed = all(r.status == ResultStatus.passed for r in all_results)
        run.status = RunStatus.passed if all_passed else RunStatus.failed
        run.finished_at = datetime.utcnow()
        db.commit()

    db.refresh(result)
    return result

def get_run(db: Session, run_id: str) -> TestRun | None:
    return db.query(TestRun).filter(TestRun.id == run_id).first()
```

// frontend/lib/api-client.ts
const BASE = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:8000/api/v1";

function getToken(): string | null {
  if (typeof window === "undefined") return null;
  return localStorage.getItem("access_token");
}

async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
  const token = getToken();
  const res = await fetch(`${BASE}${path}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...options.headers,
    },
  });
  if (!res.ok) {
    const err = await res.json().catch(() => ({}));
    throw new Error(err.detail ?? `HTTP ${res.status}`);
  }
  return res.json();
}

export const api = {
  post: <T>(path: string, body: unknown) =>
    request<T>(path, { method: "POST", body: JSON.stringify(body) }),
  get: <T>(path: string) => request<T>(path),
  patch: <T>(path: string, body: unknown) =>
    request<T>(path, { method: "PATCH", body: JSON.stringify(body) }),
};

import stripe
from core.config import settings

stripe.api_key = settings.stripe_secret_key

PLANS = {
    "free": {"runs_per_month": 50, "price_id": None},
    "pro": {"runs_per_month": 5000, "price_id": "price_pro_monthly"},
    "enterprise": {"runs_per_month": -1, "price_id": "price_enterprise_monthly"},
}

def create_checkout_session(org_id: str, plan: str, success_url: str, cancel_url: str) -> str:
    if plan not in PLANS or PLANS[plan]["price_id"] is None:
        raise ValueError(f"Plan {plan} is not billable")
    session = stripe.checkout.Session.create(
        mode="subscription",
        line_items=[{"price": PLANS[plan]["price_id"], "quantity": 1}],
        success_url=success_url,
        cancel_url=cancel_url,
        metadata={"org_id": org_id},
    )
    return session.url

def handle_webhook(payload: bytes, sig_header: str) -> dict:
    event = stripe.Webhook.construct_event(
        payload, sig_header, settings.stripe_webhook_secret
    )
    return {"type": event["type"], "data": event["data"]}
